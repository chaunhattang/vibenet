  <script>
    function switchTab(viewId, element) {
      // Toggle Active Views
      document.querySelectorAll('.view').forEach(view => view.classList.remove('active'));
      document.getElementById(viewId).classList.add('active');

      // Update Navigation Bar Buttons Active State
      document.querySelectorAll('.nav-btn').forEach(btn => btn.classList.remove('active'));
      if(element) element.classList.add('active');

      // Header Dynamic Title Adjustments
      const headerTitle = document.getElementById('header-title-text');
      if (viewId === 'profile-view') {
        headerTitle.innerText = 'Profile';
      } else {
        headerTitle.innerText = 'Babagang';
      }

      // Scroll view back to top
      document.querySelector('.view-container').scrollTop = 0;
    }

    function switchToProfileView(userName) {
      if(userName) {
        document.getElementById('profile-name-heading').innerText = userName;
      }
      const profileBtn = document.getElementById('nav-profile');
      switchTab('profile-view', profileBtn);
    }
  </script>
